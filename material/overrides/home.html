Fake data for Java and Kotlin

Datafaker is a library that generates realistic test data, such as names, addresses, phone numbers and IBANs. It has over 260 providers and supports more than 70 locales. 

<dependency>
    <groupId>net.datafaker</groupId>
    <artifactId>datafaker</artifactId>
    <version>{{ df.version }}</version>
</dependency>
implementation 'net.datafaker:datafaker:{{ df.version }}'
implementation("net.datafaker:datafaker:{{ df.version }}")

Features

Schema<String, String> schema = Schema.of(
    field("name", () -> faker.name().fullName()),
    field("email", () -> faker.internet().emailAddress()));

SqlTransformer<String> sql = new SqlTransformer.SqlTransformerBuilder<String>()
    .tableName("users")
    .dialect(SqlDialect.POSTGRES)
    .batch(3)
    .build();

sql.generate(schema, 3);
